/* ─────────────────────────────────────────────────────────

   ESTADO

   ───────────────────────────────────────────────────────── */

const state = {

  nihss: null,   // 'mild' | 'moderate' | 'severe'

  size:  null,   // 'minor' | 'moderate' | 'major'

  time:  null,

  unit:  'h'

};


/* ─────────────────────────────────────────────────────────

   LÓGICA

   ───────────────────────────────────────────────────────── */

function ehraTargetDay(nihss) {

  return nihss === 'mild' ? 3 : nihss === 'moderate' ? 6 : 12;

}


function ehraVerdict(days) {

  const target = ehraTargetDay(state.nihss);

  if (days >= target) {

    return { status: 'go', title: 'Iniciar', target };

  }

  const remaining = Math.max(1, Math.ceil(target - days));

  return { status: 'wait', title: `Día ${target}`, target, remaining };

}


/* Ventanas de ELAN según tamaño del infarto.

   Cada ventana son los límites [inicio, fin) en horas. */

const ELAN_WINDOWS = {

  minor:    { early: [0,   48],  late: [72,  96]  },  // ≤48h ; día 3–4

  moderate: { early: [0,   48],  late: [144, 192] },  // ≤48h ; día 6–7

  major:    { early: [144, 192], late: [288, 360] }   // día 6–7 ; día 12–14

};


function elanVerdict(hours, days) {

  const win = ELAN_WINDOWS[state.size];


  // Caso especial: en ictus mayor, ELAN no estudió inicio antes del día 6

  if (state.size === 'major' && hours < win.early[0]) {

    const remaining = Math.max(1, Math.ceil(6 - days));

    return { status: 'wait', title: 'Día 6', target: 6, remaining };

  }


  let arm;

  if (hours >= win.early[0] && hours < win.early[1])       arm = 'early';

  else if (hours >= win.early[1] && hours < win.late[0])   arm = 'between';

  else if (hours >= win.late[0]  && hours < win.late[1])   arm = 'late';

  else                                                     arm = 'beyond';


  return { status: 'go', title: 'Iniciar', arm };

}


function computeVerdict() {

  if (!state.nihss || !state.size || state.time === null || state.time === '' || isNaN(state.time)) {

    return { kind: 'incomplete' };

  }

  const hours = state.unit === 'd' ? Number(state.time) * 24 : Number(state.time);

  const days = hours / 24;

  return {

    kind: 'ok',

    hours,

    days,

    ehra: ehraVerdict(days),

    elan: elanVerdict(hours, days)

  };

}


/* ─────────────────────────────────────────────────────────

   RENDER VEREDICTO

   ───────────────────────────────────────────────────────── */

const verdictEl = document.getElementById('verdict');


function render() {

  const r = computeVerdict();


  if (r.kind === 'incomplete') {

    verdictEl.innerHTML = `<div class="verdict-incomplete">Completa los tres apartados para ver la comparativa.</div>`;

    return;

  }


  const divergent = r.ehra.status !== r.elan.status;


  verdictEl.innerHTML = `

    <div class="verdict-head">Recomendación</div>

    <div class="dual">

      ${cardHTML('ehra', r.ehra, ehraSubtitle(r.ehra))}

      ${cardHTML('elan', r.elan, elanSubtitle(r.elan))}

    </div>

    ${divergent ? divergenceHTML(r) : ''}

    <button class="why-btn" id="openSheet">Ver justificación y citas <span class="arrow">→</span></button>

  `;


  document.getElementById('openSheet').addEventListener('click', openSheet);

}


function cardHTML(which, v, sub) {

  const sourceLabel = which === 'ehra' ? 'EHRA 2021' : 'ELAN 2023';

  const meta = which === 'ehra' ? 'Regla 1-3-6-12 (NIHSS)' : 'Basado en imagen';

  return `

    <div class="v-card is-${which} ${v.status}">

      <div class="v-source">${sourceLabel}</div>

      <div class="v-meta">${meta}</div>

      <div class="v-title">${v.title}</div>

      <div class="v-sub">${sub}</div>

    </div>

  `;

}


function ehraSubtitle(v) {

  if (v.status === 'go') {

    return `Has superado el día ${v.target} para un ictus ${nihssWord(state.nihss)}.<br><strong>Imagen previa</strong> para descartar transformación hemorrágica.`;

  }

  return `Faltan aprox. <strong>${v.remaining} día${v.remaining === 1 ? '' : 's'}</strong> hasta el día ${v.target}.`;

}


function elanSubtitle(v) {

  if (v.status === 'wait') {

    return `Faltan aprox. <strong>${v.remaining} día${v.remaining === 1 ? '' : 's'}</strong>. ELAN no estudió el inicio antes del día 6 en ictus mayor.`;

  }

  // status === 'go'

  switch (v.arm) {

    case 'early':

      return `Encaja con la <strong>rama precoz</strong> del ensayo.`;

    case 'late':

      return `Encaja con la <strong>rama tardía</strong> del ensayo. Ambas ramas demostraron seguridad.`;

    case 'between':

      return `Entre las dos ramas estudiadas. Ambas fueron seguras, iniciar es razonable.`;

    case 'beyond':

      return `Más allá de la ventana del ensayo; iniciar es claramente seguro.`;

  }

}


function nihssWord(n) {

  return n === 'mild' ? 'leve' : n === 'moderate' ? 'moderado' : 'grave';

}


function sizeWord(s) {

  return s === 'minor' ? 'menor' : s === 'moderate' ? 'moderado' : 'mayor';

}


function divergenceHTML(r) {

  return `

    <div class="diverge">

      <strong><span class="icon">⚠</span> Discrepancia entre ambos enfoques</strong>

      ${divergenceText(r)}

    </div>

  `;

}


function divergenceText(r) {

  const ehraGo = r.ehra.status === 'go';

  const elanGo = r.elan.status === 'go';

  if (ehraGo && !elanGo) {

    return `EHRA permitiría iniciar ya por NIHSS ${nihssWord(state.nihss)}, pero ELAN obliga a esperar al día 6 porque el infarto es mayor por imagen. <strong>Prima el criterio radiológico</strong>: el riesgo de transformación hemorrágica depende del tamaño, no solo del déficit clínico.`;

  }

  if (!ehraGo && elanGo) {

    return `EHRA mandaría esperar al día ${r.ehra.target} por NIHSS ${nihssWord(state.nihss)}, pero ELAN demostró seguridad iniciando antes en infartos ${sizeWord(state.size)}es por imagen. <strong>La evidencia ELAN respalda iniciar ya</strong> en este perfil.`;

  }

  return '';

}


/* ─────────────────────────────────────────────────────────

   PANEL / BOTTOM SHEET

   ───────────────────────────────────────────────────────── */

const sheet      = document.getElementById('sheet');

const backdrop   = document.getElementById('backdrop');

const sheetBody  = document.getElementById('sheetBody');

const sheetClose = document.getElementById('sheetClose');


function openSheet() {

  const r = computeVerdict();

  if (r.kind !== 'ok') return;

  sheetBody.innerHTML = panelContent(r);

  sheetBody.scrollTop = 0;

  sheet.classList.add('open');

  backdrop.classList.add('open');

  sheet.setAttribute('aria-hidden', 'false');

  document.body.style.overflow = 'hidden';

}

function closeSheet() {

  sheet.classList.remove('open');

  backdrop.classList.remove('open');

  sheet.setAttribute('aria-hidden', 'true');

  document.body.style.overflow = '';

}

backdrop.addEventListener('click', closeSheet);

sheetClose.addEventListener('click', closeSheet);

document.addEventListener('keydown', e => {

  if (e.key === 'Escape') closeSheet();

});


/* ─────────────────────────────────────────────────────────

   CONTENIDO DEL PANEL

   ───────────────────────────────────────────────────────── */

function panelContent(r) {

  const days = r.days.toFixed(1);

  return `

    <div class="summary-box">

      <span class="label">Tu caso</span>

      <div class="row"><span class="k">Gravedad clínica</span><span class="v">NIHSS ${nihssWord(state.nihss)}</span></div>

      <div class="row"><span class="k">Tamaño en imagen</span><span class="v">${sizeWord(state.size)}</span></div>

      <div class="row"><span class="k">Tiempo de evolución</span><span class="v">${days} d (${Math.round(r.hours)} h)</span></div>

    </div>


    ${ehraSection(r)}

    ${elanSection(r)}

    ${(r.ehra.status !== r.elan.status) ? divergenceSection(r) : ''}

    ${limitationsSection()}

  `;

}

  const target = r.ehra.target;

  const v = r.ehra;

  return `

    <div class="pblock">

      <h3>EHRA 2021 <span class="badge ehra">${v.title}</span></h3>


      <p>La <strong>European Heart Rhythm Association</strong> recomienda diferir el inicio del ACOD según la gravedad clínica medida por la NIHSS. Para un ictus <strong>${nihssWord(state.nihss)}</strong>, el inicio se sitúa a partir del <strong>día ${target}</strong>.</p>


      <details class="cite">

        <summary>Cita · regla 1-3-6-12</summary>

        <div class="cite-body">

          «If infarct size is not expected to substantially increase the risk of haemorrhagic transformation in patients with mild stroke, OAC may be initiated ≥ 3 days after AIS. In patients with moderate stroke, anticoagulation may be started ≥ 6–8 days and in patients with severe stroke at ≥ 12–14 days, after excluding secondary haemorrhagic transformation by repeating brain imaging.»

          <span class="cite-source">Steffel J et al. EHRA 2021 Practical Guide, sección «AF patients post-ischaemic stroke», p. 1654</span>

        </div>

      </details>


      <p>Antes de iniciar es <strong>obligatorio descartar transformación hemorrágica</strong> con TC o RM, ya que la disrupción de la barrera hematoencefálica tras el ictus aumenta el riesgo de sangrado parenquimatoso secundario.</p>


      <details class="cite">

        <summary>Cita · imagen previa</summary>

        <div class="cite-body">

          «Since stroke-related disruption of the blood–brain barrier increases the risk of secondary haemorrhagic transformation, timing of (re-)starting OAC must balance the risk of recurrent ischaemic stroke vs. risk of parenchymal bleeding.»

          <span class="cite-source">EHRA 2021, p. 1653</span>

        </div>

      </details>


      <p>Los autores reconocen que <strong>la regla es expert-opinion</strong>, no derivada de ECA: «These time frames and actions represent expert opinion-driven practical advice until more evidence becomes available.»</p>


      <details class="cite">

        <summary>Cita · nivel de evidencia</summary>

        <div class="cite-body">

          «These time frames and actions represent expert opinion-driven practical advice until more evidence becomes available. A multidisciplinary team approach appears mandatory in these challenging situations.»

          <span class="cite-source">EHRA 2021, p. 1654 + Figura 21</span>

        </div>

      </details>


      <details class="cite">

        <summary>Cita · guías ESC 2020</summary>

        <div class="cite-body">

          «OAC should be (re-)initiated as soon as considered possible from the neurological perspective (in most cases within the first 2 weeks).»

          <span class="cite-source">Hindricks G et al. 2020 ESC Guidelines for AF, Eur Heart J 2021; 42:373-498</span>

        </div>

      </details>

    </div>

  `;

}


function elanSection(r) {

  const v = r.elan;


  // Texto contextual según en qué rama / ventana cae el caso

  let armText = '';

  if (v.status === 'go') {

    const w = ELAN_WINDOWS[state.size];

    const lateRange = state.size === 'minor' ? 'día 3–4'

                    : state.size === 'moderate' ? 'día 6–7'

                    : 'día 12–14';

    const earlyRange = state.size === 'major' ? 'día 6–7' : '≤48 h';


    switch (v.arm) {

      case 'early':

        armText = `Tu caso encaja con la <strong>rama precoz</strong> (${earlyRange}) del ensayo. Esta rama demostró menos ictus recurrentes sin aumentar la HIC sintomática.`;

        break;

      case 'late':

        armText = `Tu caso encaja con la <strong>rama tardía</strong> (${lateRange}) del ensayo. ELAN demostró que ambas ramas son seguras; ninguna mostró exceso de HIC.`;

        break;

      case 'between':

        armText = `Tu caso queda <strong>entre las dos ramas estudiadas</strong> (entre la rama precoz y la tardía). Si la rama precoz fue segura y la tardía también, no hay razón para suponer mayor riesgo en este intervalo intermedio.`;

        break;

      case 'beyond':

        armText = `Tu caso está <strong>más allá de la rama tardía</strong> del ensayo. Iniciar es claramente seguro: ambos brazos estudiados ya habrían iniciado el ACOD en este momento.`;

        break;

    }

  } else {

    armText = `En ictus <strong>mayor</strong> por imagen, ELAN no estudió el inicio antes del día 6. Ambas ramas del ensayo (precoz y tardía) iniciaron el ACOD a partir de ese punto, por lo que iniciar antes sería extrapolar más allá de la evidencia disponible.`;

  }


  return `

    <div class="pblock">

      <h3>ELAN 2023 <span class="badge elan">${v.title}</span></h3>


      <p>El ensayo <strong>ELAN</strong> (n = 2 013, NEJM 2023) estratificó por <strong>tamaño radiológico</strong>, no por NIHSS, porque la escala clínica depende también de la localización del infarto. ${armText}</p>


      <details class="cite">

        <summary>Cita · diseño del ensayo</summary>

        <div class="cite-body">

          «Early treatment was defined as initiation of a DOAC within 48 hours after stroke onset in participants with minor or moderate stroke and on day 6 or 7 in those with major stroke. Later treatment was defined as initiation of a DOAC in participants with a minor stroke on day 3 or 4 after stroke onset, in participants with a moderate stroke on day 6 or 7, and in participants with a major stroke on day 12, 13, or 14.»

          <span class="cite-source">Fischer U et al. ELAN trial, NEJM 2023; 388:2411-21. Métodos · Trial Treatment, p. 2413</span>

        </div>

      </details>


      <details class="cite">

        <summary>Cita · por qué imagen y no NIHSS</summary>

        <div class="cite-body">

          «We chose to use an imaging-based definition of stroke severity because an alternative, the NIHSS score, is dependent on both the location and size of the infarct. The size of an infarct measured by volume analysis has been related to the risk of haemorrhagic transformation.»

          <span class="cite-source">ELAN, Discusión, p. 2419</span>

        </div>

      </details>


      <p><strong>Resultados a 30 días</strong> (rama precoz vs tardía):</p>

      <div class="panel-stats">

        <div class="panel-stat-row"><span class="label">Evento primario compuesto</span><span class="val">2,9 % vs 4,1 %</span></div>

        <div class="panel-stat-row"><span class="label">Ictus isquémico recurrente</span><span class="val">1,4 % vs 2,5 %</span></div>

        <div class="panel-stat-row"><span class="label">HIC sintomática</span><span class="val">0,2 % vs 0,2 %</span></div>

        <div class="panel-stat-row"><span class="label">Sangrado mayor extracraneal</span><span class="val">0,3 % vs 0,5 %</span></div>

      </div>


      <p><strong>Beneficio Clínico Neto (Análisis Post-hoc, JAMA Netw Open 2025)</strong></p>

      <p>Al ponderar los eventos según su impacto clínico real (ictus isquémico = 1.0; HIC = 1.5 a 3.3, dada su mayor mortalidad y discapacidad), la estrategia temprana demostró un <strong>beneficio clínico neto</strong> a pesar de la heterogeneidad de los desenlaces.</p>

      <div class="panel-stats">

        <div class="panel-stat-row"><span class="label">Beneficio a 30 días</span><span class="val">Prevención 1,7 ev/100 pax (NNT 58)</span></div>

        <div class="panel-stat-row"><span class="label">Beneficio a 90 días</span><span class="val">Prevención 2,1 ev/100 pax (NNT 47)</span></div>

      </div>

      <p>Destaca una <strong>fuerte señal de beneficio clínico neto positivo</strong> de forma específica en los participantes con ictus de <strong>tamaño mayor</strong>. Los investigadores señalan que, aunque la evidencia apoya iniciar precozmente, los intervalos de confianza amplios obligan a interpretar que no se descarta por completo un efecto neutro o daño mínimo.</p>


      <details class="cite">

        <summary>Cita · Beneficio Clínico Neto (2025)</summary>

        <div class="cite-body">

          «Evaluating success by directly summing ischemic and hemorrhagic events can be misleading... Early DOAC initiation demonstrated a net clinical benefit [...] A strong signal of net clinical benefit was observed in participants with major stroke.»

          <span class="cite-source">Polymeris AA, et al. Net Benefit of Early Anticoagulation for Stroke With Atrial Fibrillation: Post Hoc Analysis of the ELAN Randomized Clinical Trial. JAMA Netw Open. 2025;8(1):e2456307</span>

        </div>

      </details>


      ${state.size === 'major' ? `

      <p>En ictus <strong>mayor</strong>, ELAN no estudió el inicio antes del día 6: ambas ramas (precoz y tardía) arrancaron el ACOD a partir de ese punto.</p>

      <details class="cite">

        <summary>Cita · ictus mayor</summary>

        <div class="cite-body">

          «Further research is needed to confirm whether persons with atrial fibrillation and severe ischaemic stroke can receive anticoagulation earlier than 6 days after symptom onset.»

          <span class="cite-source">ELAN, Discusión, p. 2419</span>

        </div>

      </details>

      ` : ''}


      <details class="cite">

        <summary>Cita · respaldo final de los autores</summary>

        <div class="cite-body">

          «Early treatment initiation can therefore be supported if indicated or if desired.»

          <span class="cite-source">ELAN, Discusión, p. 2419</span>

        </div>

      </details>

    </div>

  `;

}


function divergenceSection(r) {

  const ehraGo = r.ehra.status === 'go';

  const elanGo = r.elan.status === 'go';


  let text = '';

  if (ehraGo && !elanGo) {

    text = `<p>EHRA permitiría iniciar (NIHSS ${nihssWord(state.nihss)} → día ${r.ehra.target}), pero ELAN obliga a esperar al día 6 porque el infarto es <strong>mayor por imagen</strong>. En esta situación <strong>prima el criterio radiológico</strong>: el riesgo de transformación hemorrágica depende del tamaño del infarto, no solo del déficit clínico, y una NIHSS baja puede infravalorar el daño tisular real (p. ej., infarto extenso de localización silente).</p>

    <p>La actitud razonable es <strong>seguir la pauta de ELAN</strong> (esperar al día 6) sobre la regla EHRA, porque la regla 1-3-6-12 se basa en opinión de expertos mientras que el límite del día 6 en ictus mayor está respaldado por un ECA con 2 013 pacientes.</p>`;

  } else if (!ehraGo && elanGo) {

    text = `<p>EHRA mandaría esperar al día ${r.ehra.target} (NIHSS ${nihssWord(state.nihss)}), pero ELAN demostró que iniciar antes en infartos <strong>${sizeWord(state.size)}es</strong> por imagen es seguro (HIC sintomática 0,2 % en ambas ramas). La discrepancia surge porque una NIHSS alta puede reflejar un infarto pequeño en localización elocuente (p. ej., cápsula interna, tronco), no necesariamente un infarto grande.</p>

    <p>La evidencia de ELAN <strong>respalda iniciar ya</strong> en este perfil. Los autores son explícitos: «el inicio precoz puede respaldarse si está indicado o se desea».</p>`;

  }


  return `

    <div class="pblock">

      <h3>Discrepancia entre ambos marcos</h3>

      ${text}

      <details class="cite">

        <summary>Cita · razón del cambio de paradigma</summary>

        <div class="cite-body">

          «We chose to use an imaging-based definition of stroke severity because an alternative, the NIHSS score, is dependent on both the location and size of the infarct. […] Our data suggest that the incidence of symptomatic intracranial haemorrhage is low with early anticoagulation if imaging-based classification is used.»

          <span class="cite-source">ELAN, Discusión, p. 2419</span>

        </div>

      </details>

    </div>

  `;

}


function limitationsSection() {

  return `

    <div class="pblock">

      <h3>Limitaciones y Consideraciones Prácticas</h3>


      <p><strong>Neuroimagen de control:</strong> La regla <strong>EHRA 2021</strong> exige una imagen de control (TC o RM) para descartar transformación hemorrágica antes de iniciar la anticoagulación. En contraste, el protocolo <strong>ELAN 2023</strong> no hizo obligatoria la neuroimagen de control de forma sistemática para todos los pacientes antes de la aleatorización o inicio del fármaco, asumiendo la seguridad guiada por la clínica y el tamaño radiológico inicial, reservándola habitualmente para pacientes con deterioro neurológico.</p>


      <p><strong>EHRA 2021:</strong> la regla 1-3-6-12 es <em>expert opinion</em>, sin respaldo de ECA. La guía reconoce explícitamente que el balance entre riesgo embólico y hemorrágico no está bien definido en intervalos concretos y propone esperar más en infartos grandes.</p>


      <p><strong>ELAN 2023:</strong> NIHSS mediano bajo en la aleatorización (3, IQR 1–6), exclusión de pacientes con anticoagulación terapéutica previa y de PH1/PH2 Heidelberg, ausencia de adjudicación central del tamaño, y población predominantemente europea sin datos de etnia. No estudió el inicio antes del día 6 en ictus mayor.</p>


      <details class="cite">

        <summary>Cita · limitaciones ELAN</summary>

        <div class="cite-body">

          «The limitations of our trial are the exclusion of persons who were already receiving therapeutic anticoagulation at baseline and the low median NIHSS score at randomization. The trial also has limited statistical power to explore subgroups […]. We did not centrally adjudicate the classification of stroke severity.»

          <span class="cite-source">ELAN, Discusión, p. 2419-2420</span>

        </div>

      </details>


      <p>En todos los escenarios, la decisión debe ser <strong>individualizada</strong> integrando edad, carga de microsangrados, hipertensión, riesgo embólico (CHA₂DS₂-VASc) y antecedente de sangrado.</p>

    </div>

  `;

}


/* ─────────────────────────────────────────────────────────

   BIND DE EVENTOS

   ───────────────────────────────────────────────────────── */

document.querySelectorAll('.cards').forEach(group => {

  const field = group.dataset.field;

  group.querySelectorAll('.card').forEach(btn => {

    btn.addEventListener('click', () => {

      state[field] = btn.dataset.val;

      group.querySelectorAll('.card').forEach(b => b.setAttribute('aria-pressed', b === btn ? 'true' : 'false'));

      render();

    });

  });

});


document.querySelectorAll('.time-units button').forEach(btn => {

  btn.addEventListener('click', () => {

    state.unit = btn.dataset.val;

    document.querySelectorAll('.time-units button').forEach(b => b.setAttribute('aria-pressed', b === btn ? 'true' : 'false'));

    render();

  });

});


document.getElementById('timeInput').addEventListener('input', e => {

  const v = e.target.value;

  state.time = v === '' ? null : Number(v);

  render();

});


render();
